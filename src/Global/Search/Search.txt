class SearchManager {
    std::vector<std::unique_ptr<ISearchProvider>> providers;

public:
    void Search(const SearchQuery& query, SearchResults& results);
};


void SearchManager::Search(
    const SearchQuery& query,
    SearchResults& results)
{
    for (auto& provider : providers) {
        if (provider->IsAvailable(query)) {
            provider->Search(query, results);
            return;
        }
    }
}
